const validView = view => [view?.position, view?.target].every(v => Array.isArray(v) && v.length === 3 && v.every(Number.isFinite));
function setPerformanceView(view) {
  if (!validView(view)) return;
  setCamMode('manual');
  // Flush the previous orbit's damping before teleporting to another pose.
  const damping = controls.enableDamping; controls.enableDamping = false; controls.update();
  camera.position.fromArray(view.position); controls.target.fromArray(view.target);
  camera.near = 0.02;
  if (Number.isFinite(view.fov)) camera.fov = clampNum(view.fov, 30, 100);
  camera.updateProjectionMatrix(); controls.update(); controls.enableDamping = damping;
}
function setShareHome(options) {
  xform = { ...XFORM_DEFAULT, rotX: -90 };
  applyXform(); refreshModelUI(); setCamMode('manual');
  if (validView(options.view)) {
    setPerformanceView({ fov: 70, ...options.view });
  } else if (options.firstCamera?.length === 4 && options.firstCamera.every(row => Array.isArray(row) && row.length === 4 && row.every(Number.isFinite))) {
    const m = options.firstCamera;
    const position = new THREE.Vector3(m[0][3], m[2][3], -m[1][3]);
    const direction = new THREE.Vector3(-m[0][2], -m[2][2], m[1][2]).normalize();
    setPerformanceView({ position: position.toArray(), target: position.clone().addScaledVector(direction, 2).toArray(), fov: 70 });
  } else {
    reframe();
  }
  controls.update();
}

const engineAPI = {
  loadSplat, startSystemAudio, stopSystemAudio, resize,
  startFile: () => $('audioFile').click(),
  openSplat: () => $('file').click(),
  savePreset: () => $('savePreset').click(),

  home: setShareHome,
  getView: () => ({ position: camera.position.toArray(), target: controls.target.toArray(), fov: camera.fov }),
  setView: setPerformanceView,
  blendView(from, to, t) {
    if (!validView(from) || !validView(to) || splatLoading || rendering) return;
    const a = new THREE.Vector3().fromArray(from.position), b = new THREE.Vector3().fromArray(to.position);
    const rotation = view => new THREE.Quaternion().setFromRotationMatrix(new THREE.Matrix4().lookAt(
      new THREE.Vector3().fromArray(view.position), new THREE.Vector3().fromArray(view.target), new THREE.Vector3(0, 1, 0)));
    camera.position.lerpVectors(a, b, t); camera.quaternion.copy(rotation(from)).slerp(rotation(to), t);
    controls.target.copy(camera.position).addScaledVector(new THREE.Vector3(0, 0, -1).applyQuaternion(camera.quaternion), 2);
    camera.fov = (from.fov || 70) + ((to.fov || 70) - (from.fov || 70)) * t;
    camera.updateProjectionMatrix(); controls.update();
  },
  setNavigation(mode) {
    controls.enableDamping = false; controls.update();
    performanceNavigation = mode === 'walk' ? 'walk' : 'orbit';
    controls.enableDamping = performanceNavigation === 'orbit';
    if (performanceNavigation === 'walk') {
      if (correcting) leaveCorrect();
      enterMode('orbit'); setCamMode('manual');
    }
    syncControlsEnabled();
  },
  look(yaw, pitch) {
    if (!mesh || splatLoading || rendering || performanceNavigation !== 'walk') return;
    const rotation = new THREE.Euler().setFromQuaternion(camera.quaternion, 'YXZ');
    rotation.y += yaw; rotation.x = clampNum(rotation.x + pitch, -1.45, 1.45); rotation.z = 0;
    camera.quaternion.setFromEuler(rotation);
    const direction = new THREE.Vector3(0, 0, -1).applyQuaternion(camera.quaternion);
    controls.target.copy(camera.position).addScaledVector(direction, 2); controls.update();
  },
  move(forward, right, up) {
    if (!mesh || splatLoading || rendering || performanceNavigation !== 'walk') return;
    const ahead = camera.getWorldDirection(new THREE.Vector3()); ahead.y = 0; ahead.normalize();
    const sideways = ahead.clone().cross(camera.up).normalize();
    const delta = ahead.multiplyScalar(forward).addScaledVector(sideways, right); delta.y = up;
    camera.position.add(delta); controls.target.add(delta); controls.update();
  },
  setSensitivity(value) { $('audSensitivity').value = value; if (liveAudio) liveAudio.sensitivity = value; },
  setMix(value) { performanceMix = clampNum(value, 0, 2); pushUniforms(); },
  setBands(values) {
    for (const band of ['low', 'mid', 'high']) if (Number.isFinite(values[band])) performanceBands[band] = clampNum(values[band], 0, 2);
    pushUniforms();
  },
  setBoost(on) { if (on && (rendering || splatLoading)) return; performanceBoost = on ? 1.8 : 1; pushUniforms(); },
  setFrozen(on) {
    if (rendering || splatLoading) return;
    performanceFrozen = on ? { values: { ...V }, time: U.time.value, grain: grainFrame } : null; pushUniforms();
  },
  setBypass(value) { performanceBypass = !!value; pushUniforms(); },
  applyLook(look, controlledIds, feel) {
    performanceFrozen = null;
    performanceLook = look.id; performanceBypass = false; performanceIds = new Set(controlledIds);
    for (const id of controlledIds) { setParam(id, Number(DEFAULT[id])); if (ENV[id]) ENV[id].on = false; }
    for (const id of Object.keys(AUD)) delete AUD[id];
    for (const [id, value] of Object.entries(look.params)) setParam(id, value);
    for (const id of ['dField', 'wMode']) $(id).value = look.selects?.[id] || $(id).options[0].value;
    for (const [id, [band, depth]] of Object.entries(look.routes)) Object.assign(audOf(id), { band, depth, ...feel });
    shapeCache.clear();
    pushUniforms(); syncSliders(); refreshKeyBtns(); refreshGroups();
    focusParam(Object.keys(look.routes)[0] || 'dAmt');
  },
  setFeel(feel) { for (const route of Object.values(AUD)) Object.assign(route, feel); shapeCache.clear(); },
  setQuality(value) { setParam('dpr', value); dpr = value; resize(); },
  state: () => ({ loading: splatLoading, rendering, live: !!liveAudio?.active, liveStatus: $('audLiveStatus').textContent,
    hasAudio: !!audioEnv, audioName, audioDur, playing, levels: { ...audioLevel },
    spectrum: liveAudio?.spectrum, liveAudio, splats: $('t-splats').textContent, fps: $('t-fps').textContent,
    error: $('t-state').textContent === 'error', status: $('status').textContent,
    mix: performanceMix, bypass: performanceBypass, look: performanceLook, navigation: performanceNavigation, asset: assetLabel,
    bands: { ...performanceBands }, boosted: performanceBoost > 1, frozen: !!performanceFrozen }),
  async thumbnail() {
    if (rendering || splatLoading) throw new Error('Scene is busy');
    await spark.update({ scene, camera });
    if (rendering || splatLoading) throw new Error('Scene is busy');
    pushUniforms(); controls.update(); pushCameraUniform(); updateShaderUniforms(); updateOptics(); renderWithPost();
    const canvas = document.createElement('canvas'); canvas.width = 400; canvas.height = 240;
    canvas.getContext('2d').drawImage(renderer.domElement, 0, 0, canvas.width, canvas.height);
    return canvas.toDataURL('image/webp', 0.65);
  },
};
